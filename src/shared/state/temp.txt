n_players          = other.n_players;
round              = other.round;
map                = other.map;
players            = other.players;
current_player     = other.current_player;
current_turn_phase = other.current_turn_phase;
tribe_stack.deep_copy(other.tribe_stack);

// then restoring pointers

// restoring pointers in Area:
for (size_t i = 0; i < map.get_areas().size(); i++)
{
    Area&        area             = map.get_area(i);
    const Tribe* other_area_owner = other.map.get_area(i).get_owner_tribe();
    if (other_area_owner != nullptr)
    {
        int owner_tribe_id = other_area_owner->id;
        for (Tribe& tribe : tribe_stack.get_in_game_tribes())
        {
            if (tribe.id == owner_tribe_id)
            {
                area.set_owner_tribe(&tribe);
                break;
            }
        }
    }
}

// restore pointers in Tribe

for (size_t i = 0; i < tribe_stack.get_in_game_tribes().size(); i++)
{
    // owned_areas
    Tribe&       tribe       = tribe_stack.get_in_game_tribes().at(i);
    const Tribe& other_tribe = other.tribe_stack.get_in_game_tribes().at(i);
    tribe.get_owned_areas().clear();
    for (Area* area : other_tribe.get_owned_areas())
    {
        tribe.get_owned_areas().push_back(&(map.get_area(area->id)));
    }

    // owner

    const Player* other_owner = other_tribe.get_owner();
    if (other_owner != nullptr)
    {
        for (Player& player : players)
        {
            if (player.id == other_owner->id)
            {
                tribe.set_owner(&player);
                break;
            }
        }
    }
}

// restoring pointers in Players

for (size_t i = 0; i < players.size(); i++)
{
    // active_tribe and tribe_in_decline
    Player&       player       = players.at(i);
    const Player& other_player = other.players.at(i);

    Tribe*                    active_tribe     = nullptr;
    Tribe*                    tribe_in_decline = nullptr;
    std::pair<Tribe*, Tribe*> other_tribes     = other_player.get_tribes();
    if (other_tribes.first != nullptr)
    {
        int active_tribe_id = other_tribes.first->id;
        for (Tribe& tribe : tribe_stack.get_in_game_tribes())
        {
            if (tribe.id == active_tribe_id)
            {
                active_tribe = &tribe;
                break;
            }
        }
    }

    if (other_tribes.second != nullptr)
    {
        int tribe_in_decline_id = other_tribes.second->id;
        for (Tribe& tribe : tribe_stack.get_in_game_tribes())
        {
            if (tribe.id == tribe_in_decline_id)
            {
                tribe_in_decline = &tribe;
                break;
            }
        }
    }

    player.set_tribes(active_tribe, tribe_in_decline);
}

// restoring pointers in Game_State

if (current_player != nullptr)
{
    int current_player_id = current_player->id;
    for (Player& player : players)
    {
        if (player.id == current_player_id)
        {
            current_player = &player;
            break;
        }
    }
}
